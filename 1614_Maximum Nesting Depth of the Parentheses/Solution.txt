class Solution {
    public int maxDepth(String s) {
        int count=0;
        int max= Integer.MIN_VALUE;
        int i=0;
        
        while(s.length() > i){
            if(s.charAt(i)=='('){
                count++;
                
                max = Math.max(max, count);
            }
            else if  (s.charAt(i)==')'){
                count--;
            }
            
            max = Math.max(max, count);
            i++;
        }
        
        return max = Math.max(max, count);
    }
}
