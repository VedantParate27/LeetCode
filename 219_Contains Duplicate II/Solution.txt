class Solution {
    public boolean containsNearbyDuplicate(int[] nums, int k) {
        HashMap<Integer, Integer> table = new HashMap<>();
        int ele;
        for(int i=0; i<nums.length; i++){
            if(table.containsKey(nums[i])){
                ele = table.get(nums[i]);
                
            if(i-ele <= k)
                return true;
            }
            table.put(nums[i], i);
        }
        
        return false;
    }
}
