class Solution {
    public int lengthOfLastWord(String s) {
        s = s.trim();
        
        int lastword = s.lastIndexOf(' ');

        String word = s.substring(lastword+1, s.length());
        int count=0;

        for(int i=0; i<word.length(); i++){
            count++;
        }
        return count;
    }
}
