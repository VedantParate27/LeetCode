class Solution {
    public boolean checkValid(int[][] matrix) {
        int n=matrix.length;
        int target=n*(n+1)/2;
        int targetSquare=n*(n+1)*(2*n+1)/6;

        for(int i=0; i<n; i++){
            int rowSum=0;
            int rowSumSquare=0;
            for(int j=0; j<n; j++){
                rowSum += matrix[i][j];
                rowSumSquare += matrix[i][j]*matrix[i][j];
            }
            if(rowSum != target || rowSumSquare != targetSquare)    return false;
        }

        for(int i=0; i<n; i++){
            int colSum=0;
            int colSumSquare=0;
            for(int j=0; j<n; j++){
                colSum += matrix[j][i];
                colSumSquare += matrix[j][i]*matrix[j][i];
            }
            if(colSum != target || colSumSquare != targetSquare)    return false;
        }

        return true;
    }
}
