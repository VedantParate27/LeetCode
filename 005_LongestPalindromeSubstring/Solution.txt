class Solution {
    public int findGCD(int[] nums) {
        int min=nums[0]; int max=nums[0];

        for(int i=0; i<nums.length; i++){
            min = Math.min(nums[i], min);
            max = Math.max(nums[i], max);
        }
        return GCD(min, max);
    }

    static int GCD(int a, int b){
        if(b==0)    return a;
        return GCD(b, a%b);
    }
}
