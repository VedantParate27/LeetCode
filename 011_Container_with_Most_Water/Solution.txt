import java.util.*;

class Solution {
    public int maxArea(int[] height) {
        int Area_max=0, left, right;
        left=0;
        right=height.length -1;

        while(left<right){
            int area=((right-left)*(Math.min(height[right],height[left])));
            Area_max = Math.max(Area_max,area);

            if(height[right]<height[left]){
                right--;
            }else{
                left++;
            }
        }
        return Area_max;
    }
}
