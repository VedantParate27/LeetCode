class Solution {
    public boolean areNumbersAscending(String s) {
        int [] arr = new int[200];
        int j=0;

        for(int i=0; i<s.length(); i++){
            if(Character.isDigit(s.charAt(i))){
                int sum=0;
                
                while(i<s.length() && Character.isDigit(s.charAt(i))){
                    sum = sum*10 + s.charAt(i) - '0'; 
                    i++;
                }
                arr[j++] = sum;
            }
        }

        for(int i=1; i<j; i++){
            if(arr[i]<=arr[i-1])     return false;
        }

        return true;
    }
}
